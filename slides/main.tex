\begin{frame}{Motivation and problem}
  \begin{itemize}
    \item Quantum image pipelines (here: FRQI encoding + QHED edge detection) could help, but must fit NISQ budgets: depth, CX count, measurement bias.
    \item In FRQI, multi-controlled address logic dominates preparation. Qiskit offers different \texttt{MCXGate} synthesis modes: \texttt{noancilla} (naive) vs \texttt{v-chain}.
    \item \textbf{Question:} when \emph{only} the preparation synthesis changes for the \emph{same} FRQI state, do both resources \emph{and} application-style scores improve under matched noise?
  \end{itemize}
  \note{Emphasize this is an engineering trade-off study, not a new edge operator. The same logical state is prepared two ways; everything downstream is held fixed.}
\end{frame}

\begin{frame}{Contributions}
  \begin{itemize}
    \item \textbf{Empirical:} structural CX/depth comparison (naive vs v-chain), a noisy-simulator robustness study, and a $2\times2$ readout-mitigation proof of concept.
    \item \textbf{Main (methodological):} a transparent recipe that reports \emph{resources, fidelity, and edge scores together} on one grid -- so a gate-count headline cannot hide an inconclusive application result.
  \end{itemize}
  \note{Lead with the contribution. The methodological recipe is the part that generalizes beyond FRQI/QHED. All claims tie to versioned CSV manifests and Python scripts in the repo.}
\end{frame}

\begin{frame}{Pipeline (fixed downstream)}
  \begin{itemize}
    \item \textbf{FRQI:} intensities as $R_y$ angles on a color qubit indexed by an address register ($m=2\log_2 N$ qubits).
    \item \textbf{QHED:} FWHT-style stage + classical Sobel scoring on decoded rasters (from \texttt{src/qhed.py}).
    \item \textbf{Compared:} \texttt{MCXGate} \texttt{noancilla} (naive) vs \texttt{v-chain}; same logical state, same QHED code path.
    \item \textbf{Flow:} decode $\rho \rightarrow$ raster $\rightarrow$ FWHT $\rightarrow$ magnitude $\rightarrow$ inverse $\rightarrow$ compare to Sobel reference.
  \end{itemize}
  \note{Stress that QHED is held constant so differences isolate preparation. Qiskit names: naive maps to noancilla decomposition; v-chain is the ancilla ladder synthesis. Implementation: src/frqi.py, src/improved.py, src/qhed.py, src/metrics.py.}
\end{frame}

\begin{frame}{MCX in FRQI: how it works}
  \begin{itemize}
    \item \textbf{Multi-controlled NOT (MCX):} flips its target \emph{only} when \emph{all} control qubits are $\ket{1}$ -- a controlled-NOT with many controls.
    \item \textbf{Role here:} controls are the address bits, so MCX \emph{selects one pixel address} out of the superposition; one pixel iteration is \textbf{match $\rightarrow$ rotate color qubit $\rightarrow$ uncompute}.
    \item \textbf{\texttt{noancilla} vs \texttt{v-chain}:} same logical gate, different compiler decomposition -- naive stays narrow but explodes in CX; v-chain borrows helper qubits (ancillas) to cut CX.
  \end{itemize}
  \note{Define MCX plainly: fires only when every control is 1. In FRQI the position qubits are the controls, so the gate picks out exactly the address being encoded; X gates around it convert a 0-bit pattern into the all-ones pattern it recognizes. v-chain detail (for questions): a Toffoli-like ladder writes the ``all ones?'' answer into an ancilla, one controlled R_y fires on that flag, then the ladder is reversed to reset the ancilla. About m-2 ancillas, reused across all pixels. Trade: a few extra qubits buy far fewer CX, which is what hurts under noise.}
\end{frame}

\begin{frame}{What is measured}
  \begin{itemize}
    \item \textbf{Resources:} qubits, depth, CX after transpilation (structural study at optimization level~3).
    \item \textbf{Noise:} depolarizing density-matrix sweeps; PSNR, SSIM, fidelity (frozen bundles at optimization level~0).
    \item \textbf{Edges:} QHED vs Sobel SSIM on noisy reconstructions (paired naive vs v-chain rows).
    \item \textbf{Readout:} $2\times2$ confusion matrix inversion slice on $4\times4$ (proof of concept).
  \end{itemize}
  \note{Mention manifests in outputs/ and thesis/results\_configuration.md for reproducibility. The point of reporting all four families is the methodological contribution.}
\end{frame}

\begin{frame}{Ideal encoding \& QHED vs Sobel}
  \begin{itemize}
    \item \textbf{Encoding is lossless} in simulation: FRQI reconstruction reaches SSIM $=1$ at every test size.
    \item \textbf{QHED vs classical Sobel} agrees at $4\times4$ but diverges as resolution grows: SSIM $\approx0.16$ at $8\times8$, $\approx0.01$ at $16\times16$.
    \item This is \emph{by design} -- QHED is a Walsh--Hadamard contrast stage, not a Sobel gradient clone -- not an encoding error.
  \end{itemize}
  \note{Noiseless reference point (experiment E1). Because QHED and Sobel differ structurally, edge SSIM is read as a \emph{relative} indicator across preparation modes, not an absolute quality score---this framing matters for the noisy edge results later.}
\end{frame}

\begin{frame}{The key result: edges do not follow CX}
  \begin{itemize}
    \item Edge SSIM (QHED on noisy reconstruction vs Sobel on the clean original) is the headline application endpoint.
    \item Across eight paired noise cells, the median $\Delta$ (v-chain$-$naive) is \emph{negative}; exploratory Wilcoxon $p\approx0.078$.
    \item So the large CX saving does \textbf{not} translate into better edge maps under this noise grid.
  \end{itemize}
  \note{This is the crux: a resource win that does not carry through to the application. Edge SSIM couples FRQI noise, decoding, QHED nonlinearity, and Sobel mismatch. Exactly why resources and application metrics are reported together.}
\end{frame}

\begin{frame}{Main takeaway}
  \begin{itemize}
    \item V-chain can \textbf{substantially reduce} transpiled CX/depth for the same FRQI state at larger $N$.
    \item Under the thesis noise grid, \textbf{edge SSIM does not} show a significant naive$\rightarrow$v-chain win (exploratory Wilcoxon $p\approx0.078$ on eight paired noise cells).
    \item \textbf{Contribution / lesson:} report resources \emph{and} application metrics together; gate-count reductions are not automatic image wins on NISQ-style models.
  \end{itemize}
  \note{Honest limitation: simulator only; 16$\times$16 v-chain DM often skipped; scaled naive extrapolates one slice. Close by restating the methodological contribution.}
\end{frame}

\begin{frame}{Limitations \& future work}
  \begin{itemize}
    \item No hardware shots; mock NISQ bundle is optional evidence only.
    \item Tiny images; small statistical $n$; scaled naive baseline extrapolates from one transpiled slice.
    \item Missing v-chain noisy/edge cells at $16\times16$ (density-matrix cost); Wilcoxon uses eight paired cells only.
    \item Future: layout-aware hardware runs, richer mitigation, complete v-chain DM at $16\times16$ if memory allows.
  \end{itemize}
  \note{Align with Conclusions: under-claim relative to the resource win; honest about extrapolation and missing cells.}
\end{frame}

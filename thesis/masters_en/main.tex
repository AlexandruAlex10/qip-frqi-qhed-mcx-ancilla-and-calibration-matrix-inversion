% Robust, breakable inline references to source code "file::function"
% identifiers. \detokenize prints underscores, slashes, dots, and "::"
% literally and works in moving arguments (captions / list-of-tables).
% An \allowbreak is inserted after each path/identifier separator so long
% names wrap inside narrow table columns and do not stretch interword
% spaces in justified text. Named \srccode to avoid clashing with the
% "code" (Snippet) float in components/Format.tex (\newfloat{code}).
\makeatletter
\edef\srccode@slash{\detokenize{/}}
\edef\srccode@dot{\detokenize{.}}
\edef\srccode@colon{\detokenize{:}}
\edef\srccode@under{\detokenize{_}}
\newcommand{\srccode@scan}[1]{%
  \@tfor\srccode@c:=#1\do{%
    \srccode@c
    \ifx\srccode@c\srccode@slash\allowbreak\fi
    \ifx\srccode@c\srccode@dot\allowbreak\fi
    \ifx\srccode@c\srccode@colon\allowbreak\fi
    \ifx\srccode@c\srccode@under\allowbreak\fi
  }%
}
\DeclareRobustCommand{\srccode}[1]{%
  \texttt{\expandafter\srccode@scan\expandafter{\detokenize{#1}}}%
}
\makeatother
